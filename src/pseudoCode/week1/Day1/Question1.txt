1.1 Pet Adoption App


CLASS Pet
    ATTRIBUTES: name, age
    CONSTRUCTOR(name, age)
        SET this.name = name
        SET this.age = age
    METHOD displayDetails()
        PRINT name, age

CLASS Dog EXTENDS Pet
    ATTRIBUTES: breed
    CONSTRUCTOR(name, age, breed)
        CALL super(name, age)
        SET this.breed = breed
    METHOD displayDetails()
        PRINT "Dog: " + name + " Age: " + age + " Breed: " + breed

CLASS Cat EXTENDS Pet
    ATTRIBUTES: color
    CONSTRUCTOR(name, age, color)
        CALL super(name, age)
        SET this.color = color
    METHOD displayDetails()
        PRINT "Cat: " + name + " Age: " + age + " Color: " + color

CLASS Bird EXTENDS Pet
    ATTRIBUTES: species
    CONSTRUCTOR(name, age, species)
        CALL super(name, age)
        SET this.species = species
    METHOD displayDetails()
        PRINT "Bird: " + name + " Age: " + age + " Species: " + species
